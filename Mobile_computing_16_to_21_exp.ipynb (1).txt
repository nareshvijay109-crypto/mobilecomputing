        "spread_signal = []\n",
        "\n",
        "for bit in data_bits:\n",
        "    symbol = 1 if bit == 1 else -1\n",
        "    for chip in spreading_code:\n",
        "        spread_signal.append(symbol * chip)\n",
        "\n",
        "# Despread by correlating each group with the original code.\n",
        "despread_bits = []\n",
        "code_length = len(spreading_code)\n",
        "\n",
        "for start in range(0, len(spread_signal), code_length):\n",
        "    received_chips = spread_signal[start:start + code_length]\n",
        "    correlation = 0\n",
        "\n",
        "    for k in range(code_length):\n",
        "        correlation += received_chips[k] * spreading_code[k]\n",
        "\n",
        "    despread_bits.append(1 if correlation > 0 else 0)\n",
        "\n",
        "print(\"Original Data:\", data_bits)\n",
        "print(\"Spreading Code:\", spreading_code)\n",
        "print(\"Spread Signal:\", spread_signal)\n",

      "source": [
        "# INPUT VALUES — achievable rate (Mbps) for each user on each resource block\n",
        "# Rows represent users; columns represent OFDMA resource blocks.\n",
        "user_rates = [\n",
        "    [8, 7, 6, 5, 4, 3, 2, 1],  # User 1\n",
        "    [2, 3, 4, 5, 6, 7, 8, 9],  # User 2\n",
        "    [5, 5, 5, 5, 5, 5, 5, 5]   # User 3\n",
        "]\n",
        "\n",
        "num_users = len(user_rates)\n",
        "num_blocks = len(user_rates[0])\n",
        "\n",
        "# Round Robin: allocate resource blocks to users in turn.\n",
        "rr_throughput = [0.0] * num_users\n",
        "\n",
        "for block in range(num_blocks):\n",
        "    selected_user = block % num_users\n",
        "    rr_throughput[selected_user] += user_rates[selected_user][block]\n",
        "\n",
        "# Proportional Fair: select the user with the largest rate / past average rate.\n",
        "pf_throughput = [0.0] * num_users\n",
        "average_rate = [1.0] * num_users\n",
        "\n",
        "for block in range(num_blocks):\n",
        "    scores = []\n",
        "    for user in range(num_users):\n",
        "        scores.append(user_rates[user][block] / average_rate[user])\n",
        "\n",
        "    selected_user = scores.index(max(scores))\n",
        "    selected_rate = user_rates[selected_user][block]\n",
        "    pf_throughput[selected_user] += selected_rate\n",
        "    average_rate[selected_user] = (\n",
        "        0.5 * average_rate[selected_user] + 0.5 * selected_rate\n",
        "    )\n",
        "\n",
        "def jain_fairness(values):\n",
        "    total = sum(values)\n",
        "    sum_squares = sum(value ** 2 for value in values)\n",
        "    if sum_squares == 0:\n",
        "        return 0\n",
        "    return (total ** 2) / (len(values) * sum_squares)\n",
        "\n",
        "print(\"ROUND ROBIN SCHEDULER\")\n",
        "for user in range(num_users):\n",
        "    print(\"User\", user + 1, \"Throughput:\", round(rr_throughput[user], 2), \"Mbps\")\n",
        "print(\"Total Throughput:\", round(sum(rr_throughput), 2), \"Mbps\")\n",
        "print(\"Jain Fairness Index:\", round(jain_fairness(rr_throughput), 3))\n",
        "\n",
        "print(\"\\nPROPORTIONAL FAIR SCHEDULER\")\n",
        "for user in range(num_users):\n",
        "    print(\"User\", user + 1, \"Throughput:\", round(pf_throughput[user], 2), \"Mbps\")\n",
        "print(\"Total Throughput:\", round(sum(pf_throughput), 2), \"Mbps\")\n",
        "print(\"Jain Fairness Index:\", round(jain_fairness(pf_throughput), 3))"
